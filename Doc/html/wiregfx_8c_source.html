114/**
115 * @brief Renderiza un cubo 3D rotando sobre los ejes X, Y y Z en la pantalla OLED.
116 * @details Aplica rotaciones tridimensionales mediante la tabla trigonométrica `sin_LUT`
117 * y proyecta los vértices calculados sobre el framebuffer de memoria del SSD1306.
118 * @pre El controlador SSD1306 debe haber sido inicializado y configurado.
119 * @post Limpia el buffer con `ssd1306_Fill(Black)` y traza las 12 aristas con `ssd1306_Line`.
120 * Incrementa los ángulos de rotación `angle_x`, `angle_y` y `angle_z`.
121 * @see sin_LUT
122 * @see cube_vertices
123 * @see cube_edges
124 * @see ssd1306_Line
125 */
126void WIREGFX_Graphics_DrawCube(void) {
127 Point2D projected[8];
128 int32_t sin_x = sin_LUT[angle_x];
129 int32_t cos_x = sin_LUT[(uint8_t)(angle_x + 64)];
130 int32_t sin_y = sin_LUT[angle_y];
131 int32_t cos_y = sin_LUT[(uint8_t)(angle_y + 64)];
132 int32_t sin_z = sin_LUT[angle_z];
133 int32_t cos_z = sin_LUT[(uint8_t)(angle_z + 64)];
134
135 ssd1306_Fill(Black);
136
137 for (int i = 0; i < 8; i++) {
138 int32_t x = cube_vertices[i].x;
139 int32_t y = cube_vertices[i].y;
140 int32_t z = cube_vertices[i].z;
141
142 // Rotaciones (X, Y, Z) aplicando corrimiento >> 7 para punto fijo
143 int32_t xy = (y * cos_x - z * sin_x) >> 7;
144 int32_t xz = (y * sin_x + z * cos_x) >> 7;
145 y = xy; z = xz;
146
147 int32_t yx = (x * cos_y + z * sin_y) >> 7;
148 int32_t yz = (-x * sin_y + z * cos_y) >> 7;
149 x = yx; z = yz;
150
151 int32_t zx = (x * cos_z - y * sin_z) >> 7;
152 int32_t zy = (x * sin_z + y * cos_z) >> 7;
153 x = zx; y = zy;
154
155 projected[i].x = x + CENTER_X;
156 projected[i].y = y + CENTER_Y;
157 }
158
159 for (int i = 0; i < 12; i++) {
160 ssd1306_Line(projected[cube_edges[i][0]].x, projected[cube_edges[i][0]].y,
161 projected[cube_edges[i][1]].x, projected[cube_edges[i][1]].y, White);
162 }
163
164 angle_x += 2; angle_y += 1; angle_z += 3;
165}

167/**
168 * @brief Renderiza un teseracto (hipercubo 4D) con proyección en perspectiva estereográfica.
169 * @details Realiza rotaciones en los planos hiperespaciales X-Z, Y-Z y X-W sobre 16 vértices
170 * y 32 aristas conectadas, escalando por el factor de perspectiva del eje $W$.
171 * @pre Display SSD1306 inicializado.
172 * @post Modifica el framebuffer de video del SSD1306; avanza los ángulos `angle_xz`, `angle_yz`, `angle_xw`.
173 * @see tesseract_vertices
174 * @see tesseract_edges
175 * @see ssd1306_Line
176 */
177void WIREGFX_Graphics_DrawTesseract(void) {
178 Point2D projected[16];
179 int32_t sin_xz = sin_LUT[angle_xz];
180 int32_t cos_xz = sin_LUT[(uint8_t)(angle_xz + 64)];
181 int32_t sin_yz = sin_LUT[angle_yz];
182 int32_t cos_yz = sin_LUT[(uint8_t)(angle_yz + 64)];
183 int32_t sin_xw = sin_LUT[angle_xw];
184 int32_t cos_xw = sin_LUT[(uint8_t)(angle_xw + 64)];
185
186 ssd1306_Fill(Black);
187
188 for (int i = 0; i < 16; i++) {
189 int32_t x = tesseract_vertices[i].x;
190 int32_t y = tesseract_vertices[i].y;
191 int32_t z = tesseract_vertices[i].z;
192 int32_t w = tesseract_vertices[i].w;
193
194 int32_t xw_x = (x * cos_xw - w * sin_xw) >> 7;
195 int32_t xw_w = (x * sin_xw + w * cos_xw) >> 7;
196 x = xw_x; w = xw_w;
197
198 int32_t xz_x = (x * cos_xz - z * sin_xz) >> 7;
199 int32_t xz_z = (x * sin_xz + z * cos_xz) >> 7;
200 x = xz_x; z = xz_z;
201
202 int32_t yz_y = (y * cos_yz - z * sin_yz) >> 7;
203 int32_t yz_z = (y * sin_yz + z * cos_yz) >> 7;
204 y = yz_y; z = yz_z;
205
206 int32_t w_factor = 48 - w;
207 x = (x * w_factor) >> 6;
208 y = (y * w_factor) >> 6;
209
210 projected[i].x = x + CENTER_X;
211 projected[i].y = y + CENTER_Y;
212 }
213
214 for (int i = 0; i < 32; i++) {
215 ssd1306_Line(projected[tesseract_edges[i][0]].x, projected[tesseract_edges[i][0]].y,
216 projected[tesseract_edges[i][1]].x, projected[tesseract_edges[i][1]].y, White);
217 }
218
219 angle_xz += 1; angle_yz += 1; angle_xw += 2;
220}

338/**
339 * @brief Gestiona el ciclo secuencial de animación de figuras tridimensionales.
340 * @details Evalúa el delta de tiempo con `HAL_GetTick()`. Cada 10 segundos conmuta cíclicamente
341 * la figura geométrica: Cubo (0) &rarr; Teseracto (1) &rarr; Pirámide (2).
342 * @pre Debe invocarse periódicamente desde la tarea de interfaz de usuario (ej. ciclo de 20 ms).
343 * @post Ejecuta la función de renderizado de la figura correspondiente sobre el framebuffer del SSD1306.
344 * @see WIREGFX_Graphics_DrawCube
345 * @see WIREGFX_Graphics_DrawTesseract
346 * @see WIREGFX_Graphics_DrawPyramid
347 * @see WIREGFX_ResetCycle
348 */
349void WIREGFX_DisplayTask(void) {
350 uint32_t now = HAL_GetTick();
351 if (wiregfx_last_switch_tick == 0) {
352 wiregfx_last_switch_tick = now;
353 }
354
355 // Cada 10 segundos (10,000 ms) se cambia de figura
356 if ((now - wiregfx_last_switch_tick) >= 10000) {
357 wiregfx_last_switch_tick = now;
358 wiregfx_figure_index = (wiregfx_figure_index + 1) % 3;
359 }
360
361 // Orden de renderizado: Cubo -> Teseracto -> Pirámide
362 switch (wiregfx_figure_index) {
363 case 0:
364 WIREGFX_Graphics_DrawCube();
365 break;
366 case 1:
367 WIREGFX_Graphics_DrawTesseract();
368 break;
369 case 2:
370 WIREGFX_Graphics_DrawPyramid();
371 break;
372 default:
373 wiregfx_figure_index = 0;
374 WIREGFX_Graphics_DrawCube();
375 break;
376 }
377}
