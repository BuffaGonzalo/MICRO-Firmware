18/**
19 * @brief Estados operativos de la máquina de estados finita del driver ESP01.
20 * @details Modela secuencialmente la vida de conexión desde el reset por hardware
21 * hasta el transporte bidireccional por socket UDP o TCP.
22 * @see _sESP01Handle
23 * @see ESP01_Task
24 * @see ESP01_StateWIFI
25 * @see ESP01_StateUDPTCP
26 */
27typedef enum{
28 ESP01_NOT_INIT = -1, /*!< Driver no inicializado o en proceso de reinicio hardware (CH_PD = 0) */
29 ESP01_WIFI_DISCONNECTED, /*!< Desconectado de la red Wi-Fi; en espera de credenciales válidas */
30 ESP01_WIFI_NOT_SETED, /*!< Sin credenciales de red Wi-Fi configuradas en memoria */
31 ESP01_WIFI_CONNECTING_WIFI, /*!< Intentando asociarse al punto de acceso (`AT+CWJAP`) */
32 ESP01_WIFI_CONNECTED, /*!< Conectado exitosamente al punto de acceso Wi-Fi (`WIFI CONNECTED`) */
33 ESP01_WIFI_NEW_IP, /*!< Dirección IP asignada por DHCP obtenida (`WIFI GOT IP`) */
34 ESP01_UDPTCP_DISCONNECTED, /*!< Socket UDP/TCP cerrado o desconectado (`CLOSED`) */
35 ESP01_UDPTCP_CONNECTING, /*!< Estableciendo socket de comunicación UDP/TCP con el host remoto (`AT+CIPSTART`) */
36 ESP01_UDPTCP_CONNECTED, /*!< Socket UDP/TCP conectado y listo para transferir datos */
37 ESP01_SEND_BUSY, /*!< Transmisión de datos en progreso por la USART (esperando prompt `>` o `SEND OK`) */
38 ESP01_SEND_READY, /*!< Buffer disponible y módulo listo para recibir nuevo comando de envío */
39 ESP01_SEND_OK, /*!< Confirmación de paquete enviado exitosamente (`SEND OK`) */
40 ESP01_SEND_ERROR /*!< Fallo o error reportado en la transmisión de datos */
41} _eESP01STATUS;

65/**
66 * @brief Inicializa las estructuras internas y timers del driver ESP01.
67 * @details Configura los buffers circulares AT, asocia los punteros de bajo nivel y ejecuta un hard-reset en CH_PD.
68 *
69 * \startuml
70 * title Máquina de Estados: Driver Wi-Fi ESP-01 (ESP01_Task)
71 * [*] --> ESP01_NOT_INIT : Reset Hardware CH_PD
72 * ESP01_NOT_INIT --> ESP01_WIFI_DISCONNECTED : Responde AT (OK)
73 * ESP01_WIFI_DISCONNECTED --> ESP01_WIFI_CONNECTING_WIFI : ESP01_SetWIFI(ssid, pass)
74 * ESP01_WIFI_CONNECTING_WIFI --> ESP01_WIFI_CONNECTED : Recibe WIFI CONNECTED
75 * ESP01_WIFI_CONNECTING_WIFI --> ESP01_WIFI_DISCONNECTED : Timeout / Error de clave
76 * ESP01_WIFI_CONNECTED --> ESP01_WIFI_NEW_IP : Recibe WIFI GOT IP
77 * state Sockets {
78 * [*] --> ESP01_UDPTCP_DISCONNECTED
79 * ESP01_UDPTCP_DISCONNECTED --> ESP01_UDPTCP_CONNECTING : ESP01_StartUDP() / TCP
80 * ESP01_UDPTCP_CONNECTING --> ESP01_UDPTCP_CONNECTED : Socket Establecido (OK)
81 * ESP01_UDPTCP_CONNECTED --> ESP01_UDPTCP_DISCONNECTED : AT+CIPCLOSE / Desconexión
82 * }
83 * ESP01_WIFI_NEW_IP --> Sockets
84 * \enduml
85 *
86 * @param[in] hESP01 Puntero a la estructura de manejadores de hardware `_sESP01Handle`.
87 * @pre El puerto USART correspondiente (ej. USART1) y el pin GPIO de CH_PD deben estar configurados.
88 * @post Reinicia el chip ESP-01 por hardware y prepara los buffers circulares de comandos AT.
89 * @see _sESP01Handle
90 * @see ESP01_Task
91 */
92void ESP01_Init(_sESP01Handle *hESP01);

94/**
95 * @brief Configura las credenciales de la red Wi-Fi e inicia el proceso de asociación.
96 * @details Si existía una conexión previa activa, el driver la cierra antes de conectar
97 * al nuevo SSID. El estado puede verificarse con `ESP01_StateWIFI()`.
98 * @param[in] ssid Cadena con el identificador de red (SSID).
99 * @param[in] password Cadena con la clave WPA/WPA2 de la red.
100 * @pre Driver inicializado mediante `ESP01_Init()`.
101 * @post Carga las credenciales e inicia el envío del comando `AT+CWJAP`. El estado pasa a `ESP01_WIFI_CONNECTING_WIFI`.
102 * @see ESP01_StateWIFI
103 * @see ESP01_Task
104 */

168 * @brief Transmite un bloque de datos del buffer circular a través del socket activo.

183/**
184 * @brief Configura el módulo en modo Dual (Station + SoftAP) e inicia un servidor TCP.
185 * @details El módulo crea un punto de acceso inalámbrico propio en el canal especificado
186 * y abre un socket servidor TCP en el puerto indicado (por defecto 80) para permitir
187 * la configuración de credenciales (SSID;PASS) mediante software de terminal (Hercules).
188 * @param[in] apSSID Nombre (SSID) de la red inalámbrica que emitirá el robot.
189 * @param[in] apPass Contraseña del SoftAP (mínimo 8 caracteres, o NULL/"" para red abierta).
190 * @param[in] ch Canal de radiofrecuencia Wi-Fi (1 a 13).
191 * @param[in] enc Tipo de autenticación: 0=Abierta, 2=WPA, 3=WPA2, 4=WPA/WPA2.
192 * @param[in] port Puerto de escucha del servidor TCP (ej: 80 o 8080). Si es 0, usa 80.
193 * @pre Módulo inicializado con comandos AT operativos.
194 * @post Configura `CWMODE=3`, `CWSAP`, `CIPMUX=1` y `CIPSERVER=1,port`.
195 * @see ESP01_Task
196 */
197void ESP01_SetSoftAP(const char *apSSID, const char *apPass, uint8_t ch, uint8_t enc, uint16_t port);
