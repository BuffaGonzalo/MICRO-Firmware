68/**
69 * @brief Renderiza un cubo 3D rotando sobre los ejes X, Y y Z en la pantalla OLED.
70 * @details Aplica matrices de rotación trigonométrica computadas mediante la tabla `sin_LUT`
71 * y proyecta los 8 vértices y 12 aristas en el framebuffer del SSD1306.
72 *
73 * \startuml
74 * title Pipeline Gráfico 3D (WIREGFX_Graphics_DrawCube)
75 * start
76 * :Leer vértices locales cube_vertices[8];
77 * :Consultar sin_x, cos_x, sin_y, cos_y, sin_z, cos_z de sin_LUT[];
78 * :Borrar pantalla ssd1306_Fill(Black);
79 * while (i < 8) is (Vértice pendiente)
80 * :Rotar en X: y' = (y*cos - z*sin) >> 7, z' = (y*sin + z*cos) >> 7;
81 * :Rotar en Y: x' = (x*cos + z*sin) >> 7, z' = (-x*sin + z*cos) >> 7;
82 * :Rotar en Z: x' = (x*cos - y*sin) >> 7, y' = (x*sin + y*cos) >> 7;
83 * :Proyectar a pantalla: projected[i].x = x + 64, projected[i].y = y + 32;
84 * :i++;
85 * endwhile
86 * while (k < 12) is (Arista pendiente)
87 * :Dibujar línea ssd1306_Line(projected[a], projected[b], White);
88 * :k++;
89 * endwhile
90 * :Incrementar ángulos de rotación;
91 * stop
92 * \enduml
93 *
94 * @pre El display SSD1306 debe estar inicializado mediante `ssd1306_Init()`.
95 * @post Limpia y dibuja el nuevo cuadro sobre el framebuffer en RAM; incrementa ángulos de rotación.
96 * @see ssd1306_Line
97 * @see Point3D
98 * @see Point2D
99 */
100void WIREGFX_Graphics_DrawCube(void);

130/**
131 * @brief Gestiona el ciclo secuencial de animación de figuras tridimensionales.
132 * @details Conmuta automáticamente la figura visualizada cada 10 segundos:
133 * **Cubo 3D** &rarr; **Teseracto 4D** &rarr; **Pirámide 3D**.
134 *
135 * \startuml
136 * title Máquina de Estados: Animación WireGFX (WIREGFX_DisplayTask)
137 * [*] --> Cubo : Reset / Inicio
138 * state Cubo : Renderiza WIREGFX_Graphics_DrawCube()
139 * state Teseracto : Renderiza WIREGFX_Graphics_DrawTesseract()
140 * state Piramide : Renderiza WIREGFX_Graphics_DrawPyramid()
141 *
142 * Cubo --> Teseracto : DeltaTick >= 10000 ms
143 * Teseracto --> Piramide : DeltaTick >= 10000 ms
144 * Piramide --> Cubo : DeltaTick >= 10000 ms
145 * \enduml
146 *
147 * @pre Debe ser invocado periódicamente dentro del bucle cooperativo del sistema (ej: 20 ms).
148 * @post Actualiza el framebuffer llamando a la rutina de renderizado correspondiente.
149 * @see WIREGFX_ResetCycle
150 * @see WIREGFX_Graphics_DrawCube
151 * @see WIREGFX_Graphics_DrawTesseract
152 * @see WIREGFX_Graphics_DrawPyramid
153 */
