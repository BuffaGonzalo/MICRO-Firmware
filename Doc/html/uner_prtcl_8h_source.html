27/**
28 * @brief Unión multipropósito para conversión y serialización de tipos escalares (1, 2 y 4 bytes).
29 * @details Facilita la descomposición de enteros de 16 y 32 bits a secuencias de bytes
30 * para su transmisión directa en la carga útil de la trama binaria.
31 */
32typedef union {
33 uint32_t ui32; /*!< Entero sin signo de 32 bits */
34 int32_t i32; /*!< Entero con signo de 32 bits */

37 uint8_t ui8[4]; /*!< Vector de 4 bytes sin signo */
38 int8_t i8[4]; /*!< Vector de 4 bytes con signo */
39} _uWord;

41/**
42 * @brief Estructura de control de buffer circular para canales de comunicación (Rx/Tx).
43 * @details Encapsula la memoria física del buffer, los índices de lectura/escritura,
44 * la máscara de desbordamiento circular y las variables de estado de la MEF de recepción.
45 */
46typedef struct {
47 uint8_t *buff; /*!< Puntero al array de memoria física asignado al buffer */
48 uint8_t indexR; /*!< Índice de lectura (Read Index) en el buffer circular */
49 uint8_t indexW; /*!< Índice de escritura (Write Index) en el buffer circular */
50 uint8_t indexData; /*!< Puntero al byte inicial de datos útiles (Payload) dentro del buffer */
51 uint8_t mask; /*!< Máscara binaria para avance circular sin operación módulo ($2^N - 1$) */
52 uint8_t chk; /*!< Acumulador de Checksum por operación XOR para la trama activa */
53 uint8_t nBytes; /*!< Contador de longitud de bytes recibidos o pendientes de despacho */
54 uint8_t header; /*!< Estado actual en la máquina de estados de sincronización de cabecera */
55} _sComm;
56
57/**
58 * @brief Estados de la máquina de decodificación de tramas binarias entrantes.
59 * @details La secuencia de sincronización valida rigurosamente la secuencia de preámbulo
60 * `'U'` &rarr; `'N'` &rarr; `'E'` &rarr; `'R'` antes de evaluar la longitud y el token.
61 */
62typedef enum {
63 HEADER_U, /*!< Esperando el carácter de sincronismo inicial 'U' */
64 HEADER_N, /*!< Esperando el segundo carácter de sincronismo 'N' */
65 HEADER_E, /*!< Esperando el tercer carácter de sincronismo 'E' */
66 HEADER_R, /*!< Esperando el cuarto carácter de sincronismo 'R' */
67 NBYTES, /*!< Leyendo el byte que define la longitud de la trama */
68 TOKEN, /*!< Esperando el delimitador de comando ':' */
69 ID, /*!< Leyendo el identificador del comando (_eCmd) */
70 PAYLOAD /*!< Recibiendo el cuerpo de datos útiles y validando el checksum final */
71} _eDecode;

73/**
74 * @brief Inicializa las estructuras de comunicación circular para transmisión y recepción.
75 * @param[in,out] Rx Puntero a la estructura de recepción `_sComm`.
76 * @param[in,out] Tx Puntero a la estructura de transmisión `_sComm`.
77 * @param[in] buffRx Array de memoria física para el buffer de recepción.
78 * @param[in] buffTx Array de memoria física para el buffer de transmisión.
79 * @note Los tamaños de los buffers deben coincidir con las macros `RXBUFSIZE` y `TXBUFSIZE`.
80 */
81void unerPrtcl_Init(_sComm *Rx, _sComm *Tx, volatile uint8_t *buffRx, volatile uint8_t *buffTx);

83/**
84 * @brief Escribe el encabezado estándar del protocolo UNER en el buffer de transmisión.
85 * @details Escribe la secuencia `'U'`, `'N'`, `'E'`, `'R'`, el campo de longitud ajustada,
86 * el carácter separador `':'` y el identificador de comando `ID`. Además, inicializa
87 * el cálculo de Checksum XOR de la trama.
88 * @param[in,out] dataTx Puntero a la estructura de transmisión `_sComm`.
89 * @param[in] ID Identificador del comando a despachar (definido en `_eCmd`).
90 * @param[in] frameLength Longitud de los datos que componen el payload.
91 * @return Checksum XOR calculado hasta el byte de ID.
92 */
93uint8_t unerPrtcl_PutHeaderOnTx(_sComm *dataTx, uint8_t ID, uint8_t frameLength);
94
95/**
96 * @brief Inserta un único byte en el buffer circular de transmisión y actualiza el Checksum.
97 * @param[in,out] dataTx Puntero a la estructura de transmisión `_sComm`.
98 * @param[in] byte Valor a encolar en el buffer de salida.
99 * @return Checksum acumulado resultante tras la operación XOR.
100 */
101uint8_t unerPrtcl_PutByteOnTx(_sComm *dataTx, uint8_t byte);
102
103/**
104 * @brief Inserta una cadena de caracteres terminada en nulo (`\0`) en el buffer de transmisión.
105 * @param[in,out] dataTx Puntero a la estructura de transmisión `_sComm`.
106 * @param[in] str Cadena de caracteres ASCII a transmitir.
107 * @return Checksum acumulado resultante tras encolar la totalidad de la cadena.
108 */
109uint8_t unerPrtcl_PutStrOntx(_sComm *dataTx, const char *str);

111/**
112 * @brief Extrae un byte del buffer de recepción aplicando offsets relativos a la posición de datos.
113 * @param[in,out] dataRx Puntero a la estructura de recepción `_sComm`.
114 * @param[in] start Desplazamiento inicial a sumar sobre `indexData` antes de la lectura.
115 * @param[in] end Desplazamiento adicional a sumar sobre `indexData` tras la lectura.

118uint8_t unerPrtcl_GetByteFromRx(_sComm *dataRx, uint8_t start, uint8_t end);

120/**
121 * @brief Ejecuta la máquina de estados de decodificación de cabeceras sobre los datos recibidos.
122 * @details Procesa los bytes acumulados entre `indexR` e `indexW`. Verifica el preámbulo "UNER:",
123 * contabiliza los bytes recibidos y valida el Checksum al final del payload.
124 * @param[in,out] dataRx Puntero a la estructura de recepción `_sComm`.
125 * @retval TRUE Trama completa y válida recibida con Checksum verificado.
126 * @retval FALSE Trama incompleta, datos en proceso o error de sincronización.
127 */
128uint8_t unerPrtcl_DecodeHeader(_sComm *dataRx);
