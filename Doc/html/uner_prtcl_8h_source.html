27/**
28 * @brief Unión multipropósito para conversión y serialización de tipos escalares (1, 2 y 4 bytes).
29 * @details Facilita la descomposición de enteros de 16 y 32 bits a secuencias de bytes

32 * @see _eCmd

38 int16_t i16[2]; /*!< Vector de 2 enteros con signo de 16 bits */
39 uint8_t ui8[4]; /*!< Vector de 4 bytes sin signo directos */

43/**
44 * @brief Estructura de control de buffer circular para canales de comunicación (Rx/Tx).
45 * @details Encapsula la memoria física del buffer, los índices de lectura/escritura,
46 * la máscara de desbordamiento circular y las variables de estado de la MEF de recepción.
47 * @see _eDecode
48 * @see _uWord
49 * @see unerPrtcl_Init
50 * @see unerPrtcl_DecodeHeader
51 */
52typedef struct {
53 uint8_t *buff; /*!< Puntero al array de memoria física asignado al buffer */
54 uint8_t indexR; /*!< Índice de lectura (Read Index) en el buffer circular */
55 uint8_t indexW; /*!< Índice de escritura (Write Index) en el buffer circular */
56 uint8_t indexData; /*!< Puntero al byte inicial de datos útiles (Payload) dentro del buffer */
57 uint8_t mask; /*!< Máscara binaria para avance circular sin operación módulo ($2^N - 1$) */
58 uint8_t chk; /*!< Acumulador de Checksum por operación XOR para la trama activa */
59 uint8_t nBytes; /*!< Contador de longitud de bytes recibidos o pendientes de despacho */
60 uint8_t header; /*!< Estado actual en la máquina de estados de sincronización de cabecera (_eDecode) */
61} _sComm;

63/**
64 * @brief Estados de la máquina de decodificación de tramas binarias entrantes.
65 * @details La secuencia de sincronización valida rigurosamente la secuencia de preámbulo
66 * `'U'` &rarr; `'N'` &rarr; `'E'` &rarr; `'R'` antes de evaluar la longitud y el token.
67 * @see unerPrtcl_DecodeHeader
68 * @see _sComm
69 */
70typedef enum {
71 HEADER_U, /*!< Esperando el carácter de sincronismo inicial 'U' */
72 HEADER_N, /*!< Esperando el segundo carácter de sincronismo 'N' */
73 HEADER_E, /*!< Esperando el tercer carácter de sincronismo 'E' */
74 HEADER_R, /*!< Esperando el cuarto carácter de sincronismo 'R' */
75 NBYTES, /*!< Leyendo el byte que define la longitud de la trama */
76 TOKEN, /*!< Esperando el delimitador de comando ':' */
77 ID, /*!< Leyendo el identificador del comando (_eCmd) */
78 PAYLOAD /*!< Recibiendo el cuerpo de datos útiles y validando el checksum final */
79} _eDecode;

93/**
94 * @brief Escribe el encabezado estándar del protocolo UNER en el buffer de transmisión.
95 * @details Escribe la secuencia `'U'`, `'N'`, `'E'`, `'R'`, el campo de longitud ajustada,
96 * el carácter separador `':'` y el identificador de comando `ID`. Además, inicializa
97 * el cálculo de Checksum XOR de la trama.
98 * @param[in,out] dataTx Puntero a la estructura de transmisión `_sComm`.
99 * @param[in] ID Identificador del comando a despachar (definido en `_eCmd`).
100 * @param[in] frameLength Longitud de los datos que componen el payload.
101 * @pre El buffer `dataTx` debe contar con al menos `TXBYTES` (7) bytes libres.
102 * @post Avanza `indexW` en 7 posiciones en anillo y actualiza `dataTx->chk` con el XOR de la cabecera.
103 * @return Checksum XOR calculado hasta el byte de ID.
104 * @see unerPrtcl_PutByteOnTx
105 * @see unerPrtcl_PutStrOntx
106 */
107uint8_t unerPrtcl_PutHeaderOnTx(_sComm *dataTx, uint8_t ID, uint8_t frameLength);

109/**
110 * @brief Inserta un único byte en el buffer circular de transmisión y actualiza el Checksum.
111 * @param[in,out] dataTx Puntero a la estructura de transmisión `_sComm`.
112 * @param[in] byte Valor a encolar en el buffer de salida.
113 * @pre Debe existir al menos un byte libre entre `indexW` e `indexR`.
114 * @post Incrementa `nBytes`, avanza `indexW` y acumula `byte` en `dataTx->chk` mediante XOR.
115 * @return Checksum acumulado resultante tras la operación XOR.
116 * @see unerPrtcl_PutHeaderOnTx
117 */
118uint8_t unerPrtcl_PutByteOnTx(_sComm *dataTx, uint8_t byte);

143/**
144 * @brief Ejecuta la máquina de estados de decodificación de cabeceras sobre los datos recibidos.
145 * @details Procesa los bytes acumulados entre `indexR` e `indexW`. Verifica el preámbulo "UNER:",
146 * contabiliza los bytes recibidos y valida el Checksum al final del payload.
147 *
148 * \startuml
149 * title Diagrama de Estados: Decodificación unerPrtcl (unerPrtcl_DecodeHeader)
150 * [*] --> HEADER_U : Inicio / Reset
151 * HEADER_U --> HEADER_N : Byte == 'U'
152 * HEADER_N --> HEADER_E : Byte == 'N'
153 * HEADER_N --> HEADER_U : Byte != 'N'
154 * HEADER_E --> HEADER_R : Byte == 'E'
155 * HEADER_E --> HEADER_U : Byte != 'E'
156 * HEADER_R --> NBYTES : Byte == 'R'
157 * HEADER_R --> HEADER_U : Byte != 'R'
158 * NBYTES --> TOKEN : Guarda nBytes
159 * TOKEN --> PAYLOAD : Byte == ':'\n(Inicializa Chk XOR)
160 * TOKEN --> HEADER_U : Byte != ':'
161 *
162 * state PAYLOAD {
163 * [*] --> Recibiendo : nBytes > 0
164 * Recibiendo : Chk ^= Byte
165 * Recibiendo --> Recibiendo : nBytes > 0
166 * Recibiendo --> Validacion : nBytes == 0
167 * Validacion : ¿Byte == Chk?
168 * }
169 *
170 * Validacion --> HEADER_U : Checksum OK (retorna TRUE)
171 * Validacion --> HEADER_U : Checksum Error (retorna FALSE)
172 * \enduml
173 *
174 * @param[in,out] dataRx Puntero a la estructura de recepción `_sComm`.
175 * @pre El buffer debe contener bytes no leídos (`indexR != indexW`).
176 * @post Modifica `indexR`, `header`, `indexData` y `chk`. Si retorna TRUE, `indexData` apunta al comando.
177 * @retval 1 (TRUE) Trama completa y válida recibida con Checksum verificado.
178 * @retval 0 (FALSE) Trama incompleta, datos en proceso o error de sincronización.
179 * @see _sComm
180 * @see _eDecode
181 * @see decodeCommand
182 */
183uint8_t unerPrtcl_DecodeHeader(_sComm *dataRx);
