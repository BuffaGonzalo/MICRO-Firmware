82/**
83 * @brief Inicializa los registros fundamentales del MPU-6050 para el lazo de 200 Hz.
84 * @details Configuración aplicada sobre el sensor:
85 * - `PWR_MGMT_1` (0x6B) = 0x00: Despierta el oscilador interno.
86 * - `SMPLRT_DIV` (0x19) = 0x04: Tasa de muestreo = 1000Hz / (1 + 4) = 200 Hz (período de 5 ms).
87 * - `CONFIG` (0x1A) = 0x02: Filtro pasa-bajos DLPF ~98 Hz (retardo mínimo 2.8 ms).
88 * - `ACCEL_CONFIG` (0x1C) = 0x00: Rango de aceleración $\pm 2g$ (16384 LSB/g).
89 * - `GYRO_CONFIG` (0x1B) = 0x00: Rango de giróscopo $\pm 250^\circ/\text{s}$ (131 LSB/(°/s)).
90 * @pre El bus I2C2 debe estar inicializado y `memWrite` enlazado.
91 * @post El sensor queda activo y configurado a 200 Hz con DLPF anti-vibración a 98 Hz.
92 * @see mpu6050_WriteData
93 * @see mpu6050_Read
94 */
95void mpu6050_Init(void)
96{
97 uint8_t data;
98
99 // 1. Salir del modo de bajo consumo (modo sleep)
100 data = 0x00;
101 mpu6050_WriteData(&data, PWR_MGMT_1_REG);
102
103 // 2. Configurar Sample Rate Divider a 4 (Muestreo a 200 Hz / 5 ms sincronizado con STM32)
104 data = 0x04;
105 mpu6050_WriteData(&data, SMPLRT_DIV_REG);
106
107 // 3. Activar filtro pasa-bajos DLPF a ~98Hz para filtrar vibraciones mecánicas de los motores
108 data = 0x02;
109 mpu6050_WriteData(&data, CONFIG_REG);
110
111 // 4. Configurar acelerómetro con rango +-2g
112 data = 0x00;
113 mpu6050_WriteData(&data, ACCEL_CONFIG_REG);
114
115 // 5. Configurar giróscopo con rango +-250 deg/s
116 data = 0x00;
117 mpu6050_WriteData(&data, GYRO_CONFIG_REG);
118}

132/**
133 * @brief Máquina de estados no bloqueante para la captura de 14 bytes inerciales continuos.
134 * @details El proceso se divide en dos fases para sincronizarse con la interrupción DMA:
135 * - En Fase 1: Dispara la lectura de 14 bytes por DMA desde `ACCEL_XOUT_H_REG` (0x3B).
136 * - En Fase 2: Tras recibir la señal de `mpu6050_RxCplt`, recompone los registros:
137 * `Ax`, `Ay`, `Az`, `Temperatura`, `Gx`, `Gy`, `Gz` desplazando el byte alto 8 bits.
138 * @return 1 si los datos fueron recibidos y ensamblados exitosamente; 0 si la transferencia sigue en curso.
139 * @pre Función `memReadDMA` asociada y bus libre.
140 * @post Al completarse (retorno 1), actualiza variables estáticas `ax, ay, az, gx, gy, gz`.
141 * @see mpu6050_GetData
142 * @see mpu6050_Reset_State
143 * @see PIDTask
144 */
145char mpu6050_Read(void)
146{
147 static uint8_t Rec_Data[14];
148
149 if (*mpu6050_RxCplt || mpu_state == 1) {
150 *mpu6050_RxCplt = 0; // Reset de la bandera de recepción DMA
151 switch (mpu_state) {
152 case 1:
153 mpu_state = 2;
154 // Lanzar solicitud DMA para leer los 14 registros de golpe (Ax, Ay, Az, Temp, Gx, Gy, Gz)
155 mpu6050_ReadDataDMA(Rec_Data, 14, ACCEL_XOUT_H_REG);
156 break;
157 case 2:
158 mpu_state = 1;
159 // Combinar bytes MSB y LSB en enteros de 16 bits con signo (formato Big-Endian del sensor)
160 ax = (int16_t) (Rec_Data[0] << 8 | Rec_Data[1]);
161 ay = (int16_t) (Rec_Data[2] << 8 | Rec_Data[3]);
162 az = (int16_t) (Rec_Data[4] << 8 | Rec_Data[5]);
163
164 gx = (int16_t) (Rec_Data[8] << 8 | Rec_Data[9]);
165 gy = (int16_t) (Rec_Data[10] << 8 | Rec_Data[11]);
166 gz = (int16_t) (Rec_Data[12] << 8 | Rec_Data[13]);
167
168 return 1; // Nuevos datos disponibles
169 break;
170 }
171 }
172 return 0; // Lectura en proceso
173}
