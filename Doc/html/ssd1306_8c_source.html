149#endif

177 ssd1306_WriteCommand(0xD3); //-set display offset - CHECK

179
180 ssd1306_WriteCommand(0xD5); //--set display clock divide ratio/oscillator frequency
181 ssd1306_WriteCommand(0xF0); //--set divide ratio
182
183 ssd1306_WriteCommand(0xD9); //--set pre-charge period
184 ssd1306_WriteCommand(0x22); //
185
186 ssd1306_WriteCommand(0xDA); //--set com pins hardware configuration - CHECK
187#if (SSD1306_HEIGHT == 32)
188 ssd1306_WriteCommand(0x02);
189#elif (SSD1306_HEIGHT == 64)
190 ssd1306_WriteCommand(0x12);
191#elif (SSD1306_HEIGHT == 128)
192 ssd1306_WriteCommand(0x12);
193#else
194#error "Only 32, 64, or 128 lines of height are supported!"
195#endif

199
200 ssd1306_WriteCommand(0x8D); //--set DC-DC enable

202 ssd1306_SetDisplayOn(1); //--turn on SSD1306 panel

223/* Write the screenbuffer with changed to the screen */
224void ssd1306_UpdateScreen(void) {
225 // Write data to each page of RAM. Number of pages
226 // depends on the screen height:
227 //
228 // * 32px == 4 pages
229 // * 64px == 8 pages
230 // * 128px == 16 pages
231 for(uint8_t i = 0; i < SSD1306_HEIGHT/8; i++) {
232 ssd1306_WriteCommand(0xB0 + i); // Set the current RAM page address.
233 ssd1306_WriteCommand(0x00 + SSD1306_X_OFFSET_LOWER);
234 ssd1306_WriteCommand(0x10 + SSD1306_X_OFFSET_UPPER);
235 ssd1306_WriteData(&SSD1306_Buffer[SSD1306_WIDTH*i],SSD1306_WIDTH);
236 }
237}

240/* Write the screenbuffer with changed to the screen */
241char ssd1306_UpdateScreenDMA(void) {
242 // Write data to each page of RAM. Number of pages
243 // depends on the screen height:
244 //
245 // * 32px == 4 pages
246 // * 64px == 8 pages
247 // * 128px == 16 pages
248
249 // Only proceed if I2C is ready or we're starting a new transaction
250
251 if (*SSD1306_TxCplt || ssd1306_dma_state == 1) {
252 *SSD1306_TxCplt = 0; // Reset completion flag
253
254 switch (ssd1306_dma_state) {
255 case 1: // Set page address
256 ssd1306_WriteCommandDMA(0xB0 + ssd1306_dma_current_page);
257 ssd1306_dma_state = 2;
258 break;
259 case 2: // Set column address low nibble
260 ssd1306_WriteCommandDMA(0x00 + SSD1306_X_OFFSET_LOWER);
261 ssd1306_dma_state = 3;
262 break;
263 case 3: // Set column address high nibble
264 ssd1306_WriteCommandDMA(0x10 + SSD1306_X_OFFSET_UPPER);
265 ssd1306_dma_state = 4;
266 break;
267 case 4: // Write page data
268 ssd1306_WriteDataDMA(&SSD1306_Buffer[SSD1306_WIDTH*ssd1306_dma_current_page],SSD1306_WIDTH);
269 ssd1306_dma_current_page++;
270 ssd1306_dma_state = 1; // Start over with next page
271
272 if (ssd1306_dma_current_page > 7){//SSD1306_HEIGHT/8) {
273 ssd1306_dma_current_page = 0;
274 return 1;
275 }
276 break;
277 }
278 }
279 return 0;
280}

291/*
292 * Draw one pixel in the screenbuffer
293 * X => X Coordinate
294 * Y => Y Coordinate
295 * color => Pixel color
296 */
297void ssd1306_DrawPixel(uint8_t x, uint8_t y, SSD1306_COLOR color) {
298 if(x >= SSD1306_WIDTH || y >= SSD1306_HEIGHT) {
299 // Don't write outside the buffer
300 return;
301 }
302
303 // Draw in the right color
304 if(color == White) {
305 SSD1306_Buffer[x + (y / 8) * SSD1306_WIDTH] |= 1 << (y % 8);
306 } else {
307 SSD1306_Buffer[x + (y / 8) * SSD1306_WIDTH] &= ~(1 << (y % 8));
308 }
309}

311/*
312 * Draw 1 char to the screen buffer
313 * ch => char om weg te schrijven
314 * Font => Font waarmee we gaan schrijven
315 * color => Black or White
316 */
317char ssd1306_WriteChar(char ch, SSD1306_Font_t Font, SSD1306_COLOR color) {
318 uint32_t i, b, j;
319
320 // Check if character is valid
321 if (ch < 32 || ch > 126)
322 return 0;
323
324 // Char width is not equal to font width for proportional font
325 const uint8_t char_width = Font.char_width ? Font.char_width[ch-32] : Font.width;
326 // Check remaining space on current line
327 if (SSD1306_WIDTH < (SSD1306.CurrentX + char_width) ||
328 SSD1306_HEIGHT < (SSD1306.CurrentY + Font.height))
329 {
330 // Not enough space on current line
331 return 0;
332 }
333
334 // Use the font to write
335 for(i = 0; i < Font.height; i++) {
336 b = Font.data[(ch - 32) * Font.height + i];
337 for(j = 0; j < char_width; j++) {
338 if((b << j) & 0x8000) {
339 ssd1306_DrawPixel(SSD1306.CurrentX + j, (SSD1306.CurrentY + i), (SSD1306_COLOR) color);
340 } else {
341 ssd1306_DrawPixel(SSD1306.CurrentX + j, (SSD1306.CurrentY + i), (SSD1306_COLOR)!color);
342 }
343 }
344 }
345
346 // The current space is now taken
347 SSD1306.CurrentX += char_width;
348
349 // Return written char for validation
350 return ch;
351}

373/* Draw line by Bresenhem's algorithm */
374void ssd1306_Line(uint8_t x1, uint8_t y1, uint8_t x2, uint8_t y2, SSD1306_COLOR color) {
375 int32_t deltaX = abs(x2 - x1);
376 int32_t deltaY = abs(y2 - y1);
377 int32_t signX = ((x1 < x2) ? 1 : -1);
378 int32_t signY = ((y1 < y2) ? 1 : -1);
379 int32_t error = deltaX - deltaY;
380 int32_t error2;
381
382 ssd1306_DrawPixel(x2, y2, color);
383
384 while((x1 != x2) || (y1 != y2)) {
385 ssd1306_DrawPixel(x1, y1, color);
386 error2 = error * 2;
387 if(error2 > -deltaY) {
388 error -= deltaY;
389 x1 += signX;
390 }
391
392 if(error2 < deltaX) {
393 error += deltaX;
394 y1 += signY;
395 }
396 }
397 return;
398}

524/* Draw circle by Bresenhem's algorithm */
525void ssd1306_DrawCircle(uint8_t par_x,uint8_t par_y,uint8_t par_r,SSD1306_COLOR par_color) {
526 int32_t x = -par_r;
527 int32_t y = 0;
528 int32_t err = 2 - 2 * par_r;
529 int32_t e2;
530
531 if (par_x >= SSD1306_WIDTH || par_y >= SSD1306_HEIGHT) {
532 return;
533 }
534
535 do {
536 ssd1306_DrawPixel(par_x - x, par_y + y, par_color);
537 ssd1306_DrawPixel(par_x + x, par_y + y, par_color);
538 ssd1306_DrawPixel(par_x + x, par_y - y, par_color);
539 ssd1306_DrawPixel(par_x - x, par_y - y, par_color);
540 e2 = err;
541
542 if (e2 <= y) {
543 y++;
544 err = err + (y * 2 + 1);
545 if(-x == y && e2 <= x) {
546 e2 = 0;
547 }
548 }
549
550 if (e2 > x) {
551 x++;
552 err = err + (x * 2 + 1);
553 }
554 } while (x <= 0);
555
556 return;
557}

559/* Draw filled circle. Pixel positions calculated using Bresenham's algorithm */
560void ssd1306_FillCircle(uint8_t par_x,uint8_t par_y,uint8_t par_r,SSD1306_COLOR par_color) {
561 int32_t x = -par_r;
562 int32_t y = 0;
563 int32_t err = 2 - 2 * par_r;
564 int32_t e2;
565
566 if (par_x >= SSD1306_WIDTH || par_y >= SSD1306_HEIGHT) {
567 return;
568 }
569
570 do {
571 for (uint8_t _y = (par_y + y); _y >= (par_y - y); _y--) {
572 for (uint8_t _x = (par_x - x); _x >= (par_x + x); _x--) {
573 ssd1306_DrawPixel(_x, _y, par_color);
574 }
575 }
576
577 e2 = err;
578 if (e2 <= y) {
579 y++;
580 err = err + (y * 2 + 1);
581 if (-x == y && e2 <= x) {
582 e2 = 0;
583 }
584 }
585
586 if (e2 > x) {
587 x++;
588 err = err + (x * 2 + 1);
589 }
590 } while (x <= 0);
591
592 return;
593}

607 uint8_t x_start = ((x1<=x2) ? x1 : x2);
608 uint8_t x_end = ((x1<=x2) ? x2 : x1);
609 uint8_t y_start = ((y1<=y2) ? y1 : y2);
610 uint8_t y_end = ((y1<=y2) ? y2 : y1);
611
612 for (uint8_t y= y_start; (y<= y_end)&&(y<SSD1306_HEIGHT); y++) {
613 for (uint8_t x= x_start; (x<= x_end)&&(x<SSD1306_WIDTH); x++) {
614 ssd1306_DrawPixel(x, y, color);
615 }
616 }
617 return;

620SSD1306_Error_t ssd1306_InvertRectangle(uint8_t x1, uint8_t y1, uint8_t x2, uint8_t y2) {
621 if ((x2 >= SSD1306_WIDTH) || (y2 >= SSD1306_HEIGHT)) {
622 return SSD1306_ERR;
623 }
624 if ((x1 > x2) || (y1 > y2)) {
625 return SSD1306_ERR;
626 }
627 uint32_t i;
628 if ((y1 / 8) != (y2 / 8)) {
629 /* if rectangle doesn't lie on one 8px row */
630 for (uint32_t x = x1; x <= x2; x++) {
631 i = x + (y1 / 8) * SSD1306_WIDTH;
632 SSD1306_Buffer[i] ^= 0xFF << (y1 % 8);
633 i += SSD1306_WIDTH;
634 for (; i < x + (y2 / 8) * SSD1306_WIDTH; i += SSD1306_WIDTH) {
635 SSD1306_Buffer[i] ^= 0xFF;
636 }
637 SSD1306_Buffer[i] ^= 0xFF >> (7 - (y2 % 8));
638 }
639 } else {
640 /* if rectangle lies on one 8px row */
641 const uint8_t mask = (0xFF << (y1 % 8)) & (0xFF >> (7 - (y2 % 8)));
642 for (i = x1 + (y1 / 8) * SSD1306_WIDTH;
643 i <= (uint32_t)x2 + (y2 / 8) * SSD1306_WIDTH; i++) {
644 SSD1306_Buffer[i] ^= mask;
645 }
646 }
647 return SSD1306_OK;
648}

650/* Draw a bitmap */
651void ssd1306_DrawBitmap(uint8_t x, uint8_t y, const unsigned char* bitmap, uint8_t w, uint8_t h, SSD1306_COLOR color) {
652 int16_t byteWidth = (w + 7) / 8; // Bitmap scanline pad = whole byte
653 uint8_t byte = 0;
654
655 if (x >= SSD1306_WIDTH || y >= SSD1306_HEIGHT) {
656 return;
657 }
658
659 for (uint8_t j = 0; j < h; j++, y++) {
660 for (uint8_t i = 0; i < w; i++) {
661 if (i & 7) {
662 byte <<= 1;
663 } else {
664 byte = (*(const unsigned char *)(&bitmap[j * byteWidth + i / 8]));
665 }
666
667 if (byte & 0x80) {
668 ssd1306_DrawPixel(x + i, y, color);
669 }
670 }
671 }
672 return;
673}
