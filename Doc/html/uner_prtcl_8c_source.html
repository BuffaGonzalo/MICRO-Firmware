42/**
43 * @brief Escribe el encabezado estructurado del protocolo UNER en el buffer de transmisión.
44 * @details Secuencia inyectada:
45 * 1. Sync chars: 'U', 'N', 'E', 'R'
46 * 2. Longitud total: frameLength + 1
47 * 3. Delimitador: ':'
48 * 4. Identificador de comando: ID
49 * Calcula el Checksum preliminar mediante XOR acumulado sobre todos los campos.
50 */
51uint8_t unerPrtcl_PutHeaderOnTx(_sComm *dataTx, uint8_t ID, uint8_t frameLength)
52{
53 frameLength++;
54 dataTx->chk = 0;
55 dataTx->indexData = dataTx->indexW;
56
57 dataTx->buff[dataTx->indexW++]='U';
58 dataTx->indexW &= dataTx->mask;
59 dataTx->buff[dataTx->indexW++]='N';
60 dataTx->indexW &= dataTx->mask;
61 dataTx->buff[dataTx->indexW++]='E';
62 dataTx->indexW &= dataTx->mask;
63 dataTx->buff[dataTx->indexW++]='R';
64 dataTx->indexW &= dataTx->mask;
65 dataTx->buff[dataTx->indexW++]=frameLength;
66 dataTx->indexW &= dataTx->mask;
67 dataTx->buff[dataTx->indexW++]=':';
68 dataTx->indexW &= dataTx->mask;
69 dataTx->buff[dataTx->indexW++]=ID;
70 dataTx->indexW &= dataTx->mask;
71
72 dataTx->nBytes = TXBYTES;
73 dataTx->chk ^= ('U' ^'N' ^'E' ^'R' ^frameLength ^':'^ID) ;
74
75 return dataTx->chk;
76}

118/**
119 * @brief Decodifica secuencialmente la cabecera y el payload de tramas entrantes.
120 * @details Implementa una máquina de estados para detectar el preámbulo "UNER:".
121 * Una vez localizado el delimitador ':', extrae el payload restando bytes
122 * hasta llegar al último, el cual debe coincidir con el checksum acumulado.
123 */
124uint8_t unerPrtcl_DecodeHeader(_sComm *dataRx)
125{
126 uint8_t auxIndex=dataRx->indexW;
127
128 // Procesar todos los bytes acumulados en el buffer circular
129 while(dataRx->indexR != auxIndex){
130 switch(dataRx->header)
131 {
132 // Sincronismo 1: Detección de 'U'
133 case HEADER_U:
134 if(dataRx->buff[dataRx->indexR] == 'U'){
135 dataRx->header = HEADER_N;
136 }
137 break;
138 // Sincronismo 2: Detección de 'N'
139 case HEADER_N:
140 if(dataRx->buff[dataRx->indexR] == 'N'){
141 dataRx->header = HEADER_E;
142 }else{
143 if(dataRx->buff[dataRx->indexR] != 'U'){
144 dataRx->header = HEADER_U;
145 dataRx->indexR--; // Retroceder para no perder una 'U' contigua
146 }
147 }
148 break;
149 // Sincronismo 3: Detección de 'E'
150 case HEADER_E:
151 if(dataRx->buff[dataRx->indexR] == 'E'){
152 dataRx->header = HEADER_R;
153 }else{
154 dataRx->header = HEADER_U;
155 dataRx->indexR--;
156 }
157 break;
158 // Sincronismo 4: Detección de 'R'
159 case HEADER_R:
160 if(dataRx->buff[dataRx->indexR] == 'R'){
161 dataRx->header = NBYTES;
162 }else{
163 dataRx->header = HEADER_U;
164 dataRx->indexR--;
165 }
166 break;
167 // Lectura de la cantidad de bytes que componen el paquete
168 case NBYTES:
169 dataRx->nBytes=dataRx->buff[dataRx->indexR];
170 dataRx->header = TOKEN;
171 break;
172 // Delimitador de inicio de comando ':'
173 case TOKEN:
174 if(dataRx->buff[dataRx->indexR] == ':'){
175 dataRx->header = PAYLOAD;
176 dataRx->indexData = dataRx->indexR+1;
177 dataRx->indexData &= dataRx->mask;
178 // Inicializar acumulador de Checksum con el preámbulo validado
179 dataRx->chk = 0;
180 dataRx->chk ^= ('U' ^'N' ^'E' ^'R' ^ dataRx->nBytes ^':') ;
181 }else{
182 dataRx->header = HEADER_U;
183 dataRx->indexR--;
184 }
185 break;
186 // Recepción del payload y validación de Checksum final
187 case PAYLOAD:
188 dataRx->nBytes--;
189 if(dataRx->nBytes>0){
190 dataRx->chk ^= dataRx->buff[dataRx->indexR];
191 }else{
192 dataRx->header = HEADER_U;
193 // Comprobación de integridad
194 if(dataRx->buff[dataRx->indexR] == dataRx->chk)
195 return TRUE; // Trama completa y válida
196 }
197 break;
198 default:
199 dataRx->header = HEADER_U;
200 break;
201 }
202 dataRx->indexR++;
203 dataRx->indexR &= dataRx->mask;
204 }
205 return FALSE; // Trama incompleta
206}
