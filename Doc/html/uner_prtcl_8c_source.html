17/**
18 * @brief Inicializa los descriptores de buffer circular de comunicación.
19 * @details Asigna los arrays de memoria física, resetea índices de lectura/escritura a cero

22 * @param[out] Tx Puntero al descriptor de transmisión `_sComm`.
23 * @param[in] buffRx Array de memoria física para recepción.
24 * @param[in] buffTx Array de memoria física para transmisión.
25 * @pre Los punteros a buffer deben apuntar a bloques contiguos de tamaño potencia de 2.
26 * @post Todos los índices se resetean a cero y el parser se sitúa en `HEADER_U`.
27 * @see _sComm
28 * @see _eDecode
29 */
30void unerPrtcl_Init(_sComm *Rx, _sComm *Tx, volatile uint8_t *buffRx, volatile uint8_t *buffTx){
31 Rx->buff = (uint8_t *)buffRx;

33 Rx->indexW = 0;

50/**
51 * @brief Escribe el encabezado estructurado del protocolo UNER en el buffer de transmisión.
52 * @details Secuencia inyectada:
53 * 1. Sync chars: 'U', 'N', 'E', 'R'
54 * 2. Longitud total: frameLength + 1
55 * 3. Delimitador: ':'
56 * 4. Identificador de comando: ID
57 * Calcula el Checksum preliminar mediante XOR acumulado sobre todos los campos.
58 * @param[in,out] dataTx Puntero a la estructura de transmisión `_sComm`.
59 * @param[in] ID Identificador del comando según `_eCmd`.
60 * @param[in] frameLength Cantidad de bytes en la carga útil (sin contar cabecera).
61 * @return Checksum XOR calculado para la cabecera.
62 * @pre El buffer `dataTx` debe estar inicializado y poseer al menos 7 bytes disponibles.
63 * @post `indexW` avanza 7 bytes en el buffer circular y `dataTx->chk` contiene el XOR parcial.
64 * @see unerPrtcl_PutByteOnTx
65 * @see unerPrtcl_PutStrOntx
66 */
67uint8_t unerPrtcl_PutHeaderOnTx(_sComm *dataTx, uint8_t ID, uint8_t frameLength)
68{
69 frameLength++;
70 dataTx->chk = 0;
71 dataTx->indexData = dataTx->indexW;
72
73 dataTx->buff[dataTx->indexW++]='U';
74 dataTx->indexW &= dataTx->mask;
75 dataTx->buff[dataTx->indexW++]='N';
76 dataTx->indexW &= dataTx->mask;
77 dataTx->buff[dataTx->indexW++]='E';
78 dataTx->indexW &= dataTx->mask;
79 dataTx->buff[dataTx->indexW++]='R';
80 dataTx->indexW &= dataTx->mask;
81 dataTx->buff[dataTx->indexW++]=frameLength;
82 dataTx->indexW &= dataTx->mask;
83 dataTx->buff[dataTx->indexW++]=':';
84 dataTx->indexW &= dataTx->mask;
85 dataTx->buff[dataTx->indexW++]=ID;
86 dataTx->indexW &= dataTx->mask;
87
88 dataTx->nBytes = TXBYTES;
89 dataTx->chk ^= ('U' ^'N' ^'E' ^'R' ^frameLength ^':'^ID) ;
90
91 return dataTx->chk;
92}

94/**
95 * @brief Inserta un byte de datos en el buffer de transmisión y actualiza el Checksum.
96 * @param[in,out] dataTx Puntero a la estructura de transmisión `_sComm`.
97 * @param[in] byte Valor numérico del byte a transmitir.
98 * @return Checksum acumulado resultante tras la operación XOR con `byte`.
99 * @pre Buffer `dataTx` con espacio libre.
100 * @post `nBytes` se incrementa en 1, `indexW` avanza y `chk` acumula `byte`.
101 * @see unerPrtcl_PutHeaderOnTx
102 */
103uint8_t unerPrtcl_PutByteOnTx(_sComm *dataTx, uint8_t byte)
104{
105 dataTx->nBytes++;
106 dataTx->buff[dataTx->indexW++]=byte;
107 dataTx->indexW &= dataTx->mask;
108 dataTx->chk ^= byte;
109 return dataTx->chk;
110}

112/**
113 * @brief Inserta una cadena de texto en el buffer de salida hasta encontrar el terminador nulo.
114 * @param[in,out] dataTx Puntero a la estructura de transmisión `_sComm`.
115 * @param[in] str Cadena de caracteres a encolar.
116 * @return Checksum final acumulado tras procesar la cadena completa.
117 * @pre `str` debe ser un puntero no nulo terminado en `\0`.
118 * @post Todos los caracteres de `str` son transferidos al buffer circular actualizando el checksum.
119 * @see unerPrtcl_PutByteOnTx
120 */
121uint8_t unerPrtcl_PutStrOntx(_sComm *dataTx, const char *str)
122{
123 volatile uint8_t globalIndex=0;
124 while(str[globalIndex]){
125 dataTx->nBytes++;
126 dataTx->buff[dataTx->indexW++]=str[globalIndex];
127 dataTx->indexW &= dataTx->mask;
128 dataTx->chk ^= str[globalIndex++];
129 }
130 return dataTx->chk;
131}

133/**
134 * @brief Lee un byte del buffer de recepción aplicando índices de desplazamiento seguro.
135 * @param[in,out] dataRx Puntero a la estructura de recepción `_sComm`.
136 * @param[in] start Desplazamiento inicial antes de la lectura.

139 * @pre Trama decodificada previamente mediante `unerPrtcl_DecodeHeader`.

143uint8_t unerPrtcl_GetByteFromRx(_sComm *dataRx, uint8_t start, uint8_t end) {
144 uint8_t getByte;
145 dataRx->indexData += start;
146 dataRx->indexData &= dataRx->mask;
147 getByte = dataRx->buff[dataRx->indexData];
148 dataRx->indexData += end;
149 dataRx->indexData &= dataRx->mask;
150 return getByte;

153/**
154 * @brief Decodifica secuencialmente la cabecera y el payload de tramas entrantes.
155 * @details Implementa una máquina de estados para detectar el preámbulo "UNER:".
156 * Una vez localizado el delimitador ':', extrae el payload restando bytes
157 * hasta llegar al último, el cual debe coincidir con el checksum acumulado.
158 *
159 * \startuml
160 * title Proceso de Decodificación de Trama unerPrtcl (unerPrtcl_DecodeHeader)
161 * start
162 * :Obtener índice de escritura actual auxIndex = dataRx->indexW;
163 * while (¿Hay bytes por procesar? (indexR != auxIndex)) is (Sí)
164 * :Leer byte de entrada dataRx->buff[indexR];
165 * :Evaluar estado del protocolo (dataRx->header);
166 * :Procesar transición de cabecera ('U' -> 'N' -> 'E' -> 'R' -> ':');
167 * :Calcular Checksum XOR acumulado sobre payload;
168 * :Avanzar indexR;
169 * endwhile (Buffer vacío)
170 * if (¿Checksum coincide y payload completo?) then (Sí)
171 * #palegreen:Retornar TRUE (1);
172 * else (No)
173 * #salmon:Retornar FALSE (0);
174 * endif
175 * stop
176 * \enduml
177 *
178 * @param[in,out] dataRx Puntero a la estructura de recepción `_sComm`.
179 * @return 1 si se completó una trama con checksum válido, 0 en caso contrario.
180 * @pre Bytes disponibles en el buffer circular.
181 * @post `indexR` avanza hasta consumir la trama o los datos actuales. `indexData` apunta al comando.
182 * @see _sComm
183 * @see _eDecode
184 * @see decodeCommand
185 */
186uint8_t unerPrtcl_DecodeHeader(_sComm *dataRx)
187{
188 uint8_t auxIndex=dataRx->indexW;
189
190 // Procesar todos los bytes acumulados en el buffer circular
191 while(dataRx->indexR != auxIndex){
192 switch(dataRx->header)
193 {
194 // Sincronismo 1: Detección de 'U'
195 case HEADER_U:
196 if(dataRx->buff[dataRx->indexR] == 'U'){
197 dataRx->header = HEADER_N;
198 }
199 break;
200 // Sincronismo 2: Detección de 'N'
201 case HEADER_N:
202 if(dataRx->buff[dataRx->indexR] == 'N'){
203 dataRx->header = HEADER_E;
204 }else{
205 if(dataRx->buff[dataRx->indexR] != 'U'){
206 dataRx->header = HEADER_U;
207 dataRx->indexR--; // Retroceder para no perder una 'U' contigua
208 }
209 }
210 break;
211 // Sincronismo 3: Detección de 'E'
212 case HEADER_E:
213 if(dataRx->buff[dataRx->indexR] == 'E'){
214 dataRx->header = HEADER_R;
215 }else{
216 dataRx->header = HEADER_U;
217 dataRx->indexR--;
218 }
219 break;
220 // Sincronismo 4: Detección de 'R'
221 case HEADER_R:
222 if(dataRx->buff[dataRx->indexR] == 'R'){
223 dataRx->header = NBYTES;
224 }else{
225 dataRx->header = HEADER_U;
226 dataRx->indexR--;
227 }
228 break;
229 // Lectura de la cantidad de bytes que componen el paquete
230 case NBYTES:
231 dataRx->nBytes=dataRx->buff[dataRx->indexR];
232 dataRx->header = TOKEN;
233 break;
234 // Delimitador de inicio de comando ':'
235 case TOKEN:
236 if(dataRx->buff[dataRx->indexR] == ':'){
237 dataRx->header = PAYLOAD;
238 dataRx->indexData = dataRx->indexR+1;
239 dataRx->indexData &= dataRx->mask;
240 // Inicializar acumulador de Checksum con el preámbulo validado
241 dataRx->chk = 0;
242 dataRx->chk ^= ('U' ^'N' ^'E' ^'R' ^ dataRx->nBytes ^':') ;
243 }else{
244 dataRx->header = HEADER_U;
245 dataRx->indexR--;
246 }
247 break;
248 // Recepción del payload y validación de Checksum final
249 case PAYLOAD:
250 dataRx->nBytes--;
251 if(dataRx->nBytes>0){
252 dataRx->chk ^= dataRx->buff[dataRx->indexR];
253 }else{
254 dataRx->header = HEADER_U;
255 // Comprobación de integridad
256 if(dataRx->buff[dataRx->indexR] == dataRx->chk)
257 return TRUE; // Trama completa y válida
258 }
259 break;
260 default:
261 dataRx->header = HEADER_U;
262 break;
263 }
264 dataRx->indexR++;
265 dataRx->indexR &= dataRx->mask;
266 }
267 return FALSE; // Trama incompleta
268}
