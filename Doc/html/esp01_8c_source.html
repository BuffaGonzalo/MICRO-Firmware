172/**
173 * @brief Configura el ESP01 en modo SoftAP + inicia servidor HTTP en puerto 80
174 *
175 * El STM32 podra recibir peticiones HTTP con SSID y contraseña para luego
176 * llamar a ESP01_SetWIFI() y conectarse como Station.
177 *
178 * @param apSSID Nombre de la red WiFi a crear
179 * @param apPass Contraseña del AP (minimo 8 chars). NULL o "" para red abierta
180 * @param ch Canal WiFi (1-13)
181 * @param enc Encriptacion: 0=abierta, 2=WPA, 3=WPA2, 4=WPA/WPA2
182 */
183void ESP01_SetWebServer(const char *apSSID, const char *apPass, uint8_t ch, uint8_t enc){
184 esp01ATSate = ESP01ATIDLE;
185 esp01Flags.byte = 0;
186 esp01WebServerMode = 1;
187
188 strncpy(esp01AP_SSID, apSSID, 31);
189 esp01AP_SSID[31] = '\0';
190
191 if(apPass && apPass[0] != '\0'){
192 strncpy(esp01AP_PASS, apPass, 31);
193 esp01AP_PASS[31] = '\0';
194 } else {
195 esp01AP_PASS[0] = '\0';
196 enc = 0; /* Sin contraseña → red abierta */
197 }
198
199 itoa(ch, esp01AP_CH, 10);
200 itoa(enc, esp01AP_ENC, 10);
201
202 esp01TimeoutTask = 50;
203 esp01ATSate = ESP01ATHARDRST0;
204 esp01TriesAT = 0;
205}

314_eESP01STATUS ESP01_Send(uint8_t connID, uint8_t *buf, uint16_t irRingBuf, uint16_t length, uint16_t sizeRingBuf){
315 if(esp01Handle.WriteUSARTByte == NULL)
316 return ESP01_NOT_INIT;
317
318 if(esp01Flags.bit.UDPTCPCONNECTED == 0 && !esp01WebServerMode)
319 return ESP01_UDPTCP_DISCONNECTED;
320
321 if(esp01Flags.bit.SENDINGDATA == 0){
322 char strInt[10];
323 uint8_t l = 0;
324
325 itoa(length, strInt, 10);
326 l = strlen(strInt);
327 if(l>4 || l==0)
328 return ESP01_SEND_ERROR;
329
330 ESP01StrToBufTX(ATCIPSEND);
331
332 /* Con CIPMUX=1 (webserver) el formato es AT+CIPSEND=connID,length */
333 /* Con CIPMUX=0 (UDP/TCP) el formato es AT+CIPSEND=length */
334 if(esp01WebServerMode){
335 char connStr[4];
336 itoa(connID, connStr, 10);
337 ESP01StrToBufTX(connStr);
338 ESP01ByteToBufTX(',');
339 }
340
341 ESP01StrToBufTX(strInt);
342 ESP01StrToBufTX("\r>");
343
344 for(uint16_t i=0; i<length; i++){
345 esp01TXATBuf[esp01iwTX++] = buf[irRingBuf++];
346 if(esp01iwTX == ESP01TXBUFAT)
347 esp01iwTX = 0;
348 if(irRingBuf == sizeRingBuf)
349 irRingBuf = 0;
350 }
351
352 esp01Flags.bit.TXCIPSEND = 1;
353 esp01Flags.bit.SENDINGDATA = 1;
354
355 if(ESP01DbgStr != NULL){
356 ESP01DbgStr("+&DBGSENDING DATA ");
357 ESP01DbgStr(strInt);
358 ESP01DbgStr("\n");
359 }
360
361 return ESP01_SEND_READY;
362 }
363
364 if(ESP01DbgStr != NULL)
365 ESP01DbgStr("+&DBGSENDING DATA BUSY\n");
366
367 return ESP01_SEND_BUSY;
368}

391 esp01TimeoutTask--;

394 esp01TimeoutDataRx--;
395 if(!esp01TimeoutDataRx)
396 esp01HState = 0;

443static void ESP01ATDecode(){
444 uint16_t i;
445 uint8_t value;
446
447 if(esp01ATSate==ESP01ATHARDRST0 || esp01ATSate==ESP01ATHARDRST1 ||
448 esp01ATSate==ESP01ATHARDRSTSTOP){
449 esp01irRXAT = esp01iwRXAT;
450 return;
451 }
452
453
454 i = esp01iwRXAT;
455 esp01TimeoutDataRx = 2;
456 while(esp01irRXAT != i){
457 value = esp01RXATBuf[esp01irRXAT];
458 switch(esp01HState){
459 case 0:
460 indexResponse = 0;
461 indexResponseChar = 4;
462 while(responses[indexResponse] != NULL){
463 if(value == responses[indexResponse][indexResponseChar]){
464 esp01nBytes = (responses[indexResponse][0] - '0');
465 esp01nBytes *= 10;
466 esp01nBytes += (responses[indexResponse][1] - '0');
467 esp01nBytes--;
468 break;
469 }
470 indexResponse++;
471 }
472 if(responses[indexResponse] != NULL){
473 esp01HState = 1;
474 indexResponseChar++;
475 }
476 else{
477 esp01TimeoutDataRx = 0;
478 if(esp01Flags.bit.WAITINGSYMBOL){
479 if(value == '>'){
480 esp01Flags.bit.WAITINGSYMBOL = 0;
481 esp01TimeoutTxSymbol = 0;
482 esp01TimeoutSendOk = 300;
483 }
484 }
485 }
486 break;
487 case 1:
488 if(value == responses[indexResponse][indexResponseChar]){
489 esp01nBytes--;
490 if(!esp01nBytes || value=='\r'){
491 esp01HState = (responses[indexResponse][2] - '0');
492 esp01HState *= 10;
493 esp01HState += (responses[indexResponse][3] - '0');
494 break;
495 }
496 }
497 else{
498 indexResponse = 0;
499 while(responses[indexResponse] != NULL){
500 esp01nBytes = (responses[indexResponse][0] - '0');
501 esp01nBytes *= 10;
502 esp01nBytes += (responses[indexResponse][1] - '0');
503 esp01nBytes -= (indexResponseChar-3);
504 if(esp01nBytes<128 && value==responses[indexResponse][indexResponseChar]){
505 if(esp01nBytes == 0){
506 esp01HState = (responses[indexResponse][2] - '0');
507 esp01HState *= 10;
508 esp01HState += (responses[indexResponse][3] - '0');
509 }
510 break;
511 }
512 indexResponse++;
513 }
514 if(responses[indexResponse] == NULL){
515 esp01HState = 0;
516 esp01irRXAT--;
517 break;
518 }
519 }
520 indexResponseChar++;
521 break;
522 case 2:
523 if(value == '\n'){
524 esp01HState = 0;
525 switch(indexResponse){
526 case 0://AT
527 case 1:
528 break;
529 case 2://OK
530 if (esp01ATSate == ESP01ATRESPONSE
531 || esp01ATSate == ESP01CWMODERESPONSE
532 || esp01ATSate == ESP01CWAUTOCONNRESPONSE
533 || esp01ATSate == ESP01CIPMUXRESPONSE
534 || esp01ATSate == ESP01CWSAPRESPONSE
535 || esp01ATSate == ESP01CWDHCPRESPONSE
536 || esp01ATSate == ESP01CIPSERVERRESPONSE
537 || esp01ATSate == ESP01CIPSTARTRESPONSE) {
538 esp01TimeoutTask = 0;
539 esp01Flags.bit.ATRESPONSEOK = 1;
540 }
541 break;
542 case 3://ERROR
543 if(esp01Flags.bit.SENDINGDATA){
544 esp01Flags.bit.SENDINGDATA = 0;
545 esp01Flags.bit.WAITINGSYMBOL = 0;
546 esp01irTX = esp01iwTX;
547 esp01TimeoutSendOk = 0;
548 }
549 break;
550 case 4://WIFI GOT IP
551 /* SOLO aceptamos el éxito si estábamos esperando la respuesta de nuestro CWJAP */
552 if (esp01ATSate == ESP01CWJAPRESPONSE) {
553 esp01TimeoutTask = 0;
554 esp01Flags.bit.ATRESPONSEOK = 1;
555 esp01Flags.bit.WIFICONNECTED = 1;
556 if (ESP01ChangeState != NULL)
557 ESP01ChangeState(ESP01_WIFI_CONNECTED);
558 }
559 break;
560 case 5://WIFI CONNECTED
561 break;
562 case 6: //WIFI DISCONNECT
563 case 7: //WIFI DISCONNECTED
564 esp01Flags.bit.UDPTCPCONNECTED = 0;
565 esp01Flags.bit.WIFICONNECTED = 0;
566
567 /* Si el evento ocurre durante la configuración AT, lo ignoramos
568 * por completo para evitar abortar el escaneo. */
569 if (esp01ATSate <= ESP01CWJAPRESPONSE
570 || esp01ATSate >= ESP01ATHARDRST0) {
571 break;
572 }
573
574 /* Si estábamos verdaderamente conectados y se cayó la red, notificamos a main */
575 if (ESP01ChangeState != NULL)
576 ESP01ChangeState(ESP01_WIFI_DISCONNECTED);
577
578 if (!esp01WebServerMode) {
579 esp01ATSate = ESP01ATHARDRSTSTOP;
580 }
581 break;
582 case 8://DISCONNECTED
583 esp01Flags.bit.UDPTCPCONNECTED = 0;
584 break;
585 case 9://SEND OK
586 esp01Flags.bit.SENDINGDATA = 0;
587 esp01TimeoutSendOk = 0;
588 if(esp01WebServerMode){
589 char connStr[4];
590 itoa(esp01LastConnID, connStr, 10);
591 ESP01StrToBufTX("AT+CIPCLOSE=");
592 ESP01StrToBufTX(connStr);
593 ESP01StrToBufTX("\r\n");
594 }
595 if(ESP01ChangeState != NULL)
596 ESP01ChangeState(ESP01_SEND_OK);
597 break;
598 case 10://CONNECT
599 /* En CIPMUX=0: "CONNECT\r\n" indica conexion UDP/TCP establecida.
600 * En CIPMUX=1: "X,CONNECT\r\n" (con ID) → no matchea aqui.
601 * Solo actuamos si NO estamos en webserver para no interferir. */
602 if(!esp01WebServerMode){
603 esp01TimeoutTask = 0;
604 esp01Flags.bit.ATRESPONSEOK = 1;
605 esp01Flags.bit.UDPTCPCONNECTED = 1;
606 if(ESP01ChangeState != NULL)
607 ESP01ChangeState(ESP01_UDPTCP_CONNECTED);
608 }
609 break;
610 case 11://CLOSED
611 esp01Flags.bit.UDPTCPCONNECTED = 0;
612 break;
613 case 13://busy
614 esp01Flags.bit.UDPTCPCONNECTED = 0;
615 esp01Flags.bit.WIFICONNECTED = 0;
616 break;
617 case 15://ready
618 esp01Flags.bit.UDPTCPCONNECTED = 0;
619 esp01Flags.bit.WIFICONNECTED = 0;
620 esp01ATSate = ESP01ATHARDRSTSTOP;
621 break;
622 case 16: //busy p
623 case 17: //busy s
624 if (esp01Flags.bit.SENDINGDATA) {
625 esp01Flags.bit.SENDINGDATA = 0;
626 esp01Flags.bit.WAITINGSYMBOL = 0;
627 esp01irTX = esp01iwTX;
628 }
629 /* Si el ESP está "ocupado" internamente con una red vieja, abortamos la
630 * espera de 15 segundos y forzamos a la librería a reintentar al instante. */
631 if (esp01ATSate > ESP01ATIDLE
632 && esp01ATSate <= ESP01ATCIFSR) {
633 esp01TimeoutTask = 0;
634 }
635 break;
636 }
637 }
638 break;
639 case 5://CIFR,STAIP
640 if(value == ','){
641 esp01HState = 6;
642 if(ESP01DbgStr != NULL)
643 ESP01DbgStr("+&DBGRESPONSE CIFSR\n");
644 }
645 else{
646 esp01HState = 0;
647 esp01irRXAT--;
648 if(ESP01DbgStr != NULL)
649 ESP01DbgStr("+&DBGERROR CIFSR 5\n");
650 }
651 break;
652 case 6:
653 if(value == '\"'){
654 esp01HState = 7;
655 esp01nBytes = 0;
656 }
657 break;
658 case 7:
659 if(value == '\"' || esp01nBytes==16)
660 esp01HState = 8;
661 else
662 esp01LocalIP[esp01nBytes++] = value;
663 break;
664 case 8:
665 if(value == '\n'){
666 esp01HState = 0;
667 if(esp01nBytes < 16){
668 esp01LocalIP[esp01nBytes] = '\0';
669 esp01Flags.bit.ATRESPONSEOK = 1;
670 esp01TimeoutTask = 0;
671 }
672 else
673 esp01LocalIP[0] = '\0';
674 if(ESP01ChangeState != NULL)
675 ESP01ChangeState(ESP01_WIFI_NEW_IP);
676 }
677 break;
678 case 10://IPD - primera coma
679 if(value == ','){
680 if(esp01WebServerMode){
681 /* Con CIPMUX=1: +IPD,connID,N:data → leer connID primero */
682 esp01HState = 13;
683 esp01LastConnID = 0;
684 } else {
685 /* Con CIPMUX=0: +IPD,N:data → leer longitud directamente */
686 esp01HState = 11;
687 esp01nBytes = 0;
688 }
689 }
690 else{
691 esp01HState = 0;
692 esp01irRXAT--;
693 }
694 break;
695 case 13://IPD connID (solo CIPMUX=1): espera la segunda coma
696 if(value == ','){
697 esp01HState = 11;
698 esp01nBytes = 0;
699 }
700 else if(value >= '0' && value <= '9'){
701 esp01LastConnID = (uint8_t)(esp01LastConnID * 10 + (value - '0'));
702 }
703 else{
704 esp01HState = 0;
705 esp01irRXAT--;
706 }
707 break;
708 case 11:
709 if(value == ':')
710 esp01HState = 12;
711 else{
712 if(value<'0' || value>'9'){
713 esp01HState = 0;
714 esp01irRXAT--;
715 }
716 else{
717 esp01nBytes *= 10;
718 esp01nBytes += (value - '0');
719 }
720 }
721 break;
722 case 12:
723 if(esp01Handle.WriteByteToBufRX != NULL)
724 esp01Handle.WriteByteToBufRX(value);
725 esp01nBytes--;
726 if(!esp01nBytes){
727 esp01HState = 0;
728 if(ESP01DbgStr != NULL)
729 ESP01DbgStr("+&DBGRESPONSE IPD\n");
730 }
731 break;
732 default:
733 esp01HState = 0;
734 esp01TimeoutDataRx = 0;
735 }
736
737 esp01irRXAT++;
738 if(esp01irRXAT == ESP01RXBUFAT)
739 esp01irRXAT = 0;
740 }
