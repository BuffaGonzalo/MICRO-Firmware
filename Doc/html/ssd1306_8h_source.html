160/**
161 * @brief Ejecuta el volcado asíncrono no bloqueante del framebuffer vía transferencias DMA.
162 * @details Implementa una máquina de 4 estados para enviar las 8 páginas (1024 bytes)
163 * de forma segmentada intercalando comandos de direccionamiento y transferencias DMA de 128 bytes.
164 *
165 * \startuml
166 * title Pipeline de Refresco No Bloqueante DMA OLED (ssd1306_UpdateScreenDMA)
167 * start
168 * if (¿Flag TxCplt activo O estado == 1?) then (Sí)
169 * :Limpiar bandera *SSD1306_TxCplt = 0;
170 * if (estado == 1) then
171 * :Comando DMA: Página (0xB0 + page);
172 * :estado = 2;
173 * elseif (estado == 2) then
174 * :Comando DMA: Columna baja (0x00 + offset);
175 * :estado = 3;
176 * elseif (estado == 3) then
177 * :Comando DMA: Columna alta (0x10 + offset);
178 * :estado = 4;
179 * elseif (estado == 4) then
180 * :Escribir 128 bytes por DMA de la página actual;
181 * :page++;
182 * :estado = 1;
183 * if (¿page > 7?) then (Fin de frame)
184 * :page = 0;
185 * #palegreen:Retornar 1 (Frame completado);
186 * stop
187 * endif
188 * endif
189 * endif
190 * #salmon:Retornar 0 (Transferencia en progreso);
191 * stop
192 * \enduml
193 *
194 * @return 1 cuando se transmitieron las 8 páginas completas (nuevo cuadro en OLED); 0 mientras continúe transfiriendo.
195 * @pre Callback de DMA Tx configurado mediante `ssd1306_ADC_ConfCpltCallback`.
196 * @post Actualiza `ssd1306_dma_state` y `ssd1306_dma_current_page`.
197 * @see ssd1306_ResetDMAState
198 */
199char ssd1306_UpdateScreenDMA(void);
