\begin{DoxyItemize}

\item CLKEN bit in the USART\+\_\+\+CR2 register,
\item SCEN bit in the USART\+\_\+\+CR3 register,
\item IREN bit in the USART\+\_\+\+CR3 register,
\item HDSEL bit in the USART\+\_\+\+CR3 register. 
\end{DoxyItemize}
